        // ================================================================

        const STORAGE_KEY = 'sop_daily_data_v2';

        function getDefaultDay() {
            return {
                // 平台执行
                replied: false,
                visitor: false,
                customer: false,
                inquiryCount: 0,
                visitorCount: 0,
                // 产品工作
                optimize: false,
                optimizeProduct: '',
                upload: false,
                uploadProduct: '',
                // 社媒
                social: false,
                socialCount: 0,
                platforms: [], // ['fans', 'linkedin', 'facebook', 'instagram']
                // 客户维护
                oldclient: false,
                oldClientName: '',
                develop: false,
                developCount: 0,
                // 生活
                wakeTime: '',
                weight: '',
                water: '',
                bowel: '',
                exercise: '',
                learning: '',
                win: '',
                improve: '',
                saved: false,
                date: ''
            };
        }

        function getAllData() {
            try {
                const raw = localStorage.getItem(STORAGE_KEY);
                return raw ? JSON.parse(raw) : {};
            } catch { return {}; }
        }

        function saveDayData(date, data) {
            const all = getAllData();
            all[date] = data;
            localStorage.setItem(STORAGE_KEY, JSON.stringify(all));
        }

        function getDayData(date) {
            const all = getAllData();
            const def = getDefaultDay();
            if (all[date]) {
                return { ...def, ...all[date], date: date };
            }
            def.date = date;
            return def;
        }

        // ================================================================
        //  UI 操作
        // ================================================================
        const todayStr = () => new Date().toISOString().slice(0, 10);
        const $ = id => document.getElementById(id);

        // 获取所有 toggle 开关的值
        function getToggles() {
            const toggles = document.querySelectorAll('.toggle-item input[type="checkbox"]');
            const result = {};
            toggles.forEach(el => {
                result[el.dataset.key] = el.checked;
            });
            return result;
        }

        // 设置 toggle 开关
        function setToggles(data) {
            const toggles = document.querySelectorAll('.toggle-item input[type="checkbox"]');
            toggles.forEach(el => {
                const key = el.dataset.key;
                if (key in data) {
                    el.checked = data[key];
                    const parent = el.closest('.toggle-item');
                    if (el.checked) {
                        parent.classList.add('checked');
                    } else {
                        parent.classList.remove('checked');
                    }
                }
            });
        }

        // 点击切换
        function toggleCheck(el) {
            const checkbox = el.querySelector('input[type="checkbox"]');
            if (checkbox) {
                checkbox.checked = !checkbox.checked;
                if (checkbox.checked) {
                    el.classList.add('checked');
                } else {
                    el.classList.remove('checked');
                }
            }
        }

        // 平台标签
        function getSelectedPlatforms() {
            const tags = document.querySelectorAll('.platform-tag.selected');
            return Array.from(tags).map(t => t.dataset.platform);
        }

            const tags = document.querySelectorAll('.platform-tag');
            tags.forEach(t => {
                t.classList.remove('selected');
                if (platforms && platforms.includes(t.dataset.platform)) {
                    t.classList.add('selected');
                }
            });
        }

        // 平台标签点击

        // ================================================================
        //  周汇总
        // ================================================================
        function getWeekRange(dateStr) {
            const d = new Date(dateStr + 'T00:00:00');
            const day = d.getDay();
            const diff = d.getDate() - day + (day === 0 ? -6 : 1);
            const monday = new Date(d);
            monday.setDate(diff);
            const sunday = new Date(monday);
            sunday.setDate(monday.getDate() + 6);
            const fmt = (dt) => dt.toISOString().slice(0, 10);
            return { start: fmt(monday), end: fmt(sunday) };
        }

        function updateWeekInfo(dateStr) {
            const range = getWeekRange(dateStr);
            $('weekInfo').textContent = `📅 ${range.start} ~ ${range.end}`;
            renderWeekSummary(range.start, range.end);
        }

        function renderWeekSummary(start, end) {
            const all = getAllData();
            const days = [];
            let current = new Date(start + 'T00:00:00');
            const endDate = new Date(end + 'T00:00:00');
            while (current <= endDate) {
                const key = current.toISOString().slice(0, 10);
                if (all[key]) {
                    days.push({ date: key, data: all[key] });
                } else {
                    days.push({ date: key, data: null });
                }
                current.setDate(current.getDate() + 1);
            }

            // 统计
            let totalDays = 0,
                replied = 0,
                visitor = 0,
                optimize = 0,
                upload = 0,
                social = 0,
                oldclient = 0,
                develop = 0;
            let repliedCount = 0,
                visitorCount = 0,
                socialCount = 0,
                developCount = 0;

            days.forEach(d => {
                if (d.data && d.data.saved) {
                    totalDays++;
                    if (d.data.replied) replied++;
                    if (d.data.visitor) visitor++;
                    if (d.data.optimize) optimize++;
                    if (d.data.upload) upload++;
                    if (d.data.social) social++;
                    if (d.data.oldclient) oldclient++;
                    if (d.data.develop) develop++;
                    repliedCount += d.data.inquiryCount || 0;
                    visitorCount += d.data.visitorCount || 0;
                    socialCount += d.data.socialCount || 0;
                    developCount += d.data.developCount || 0;
                }
            });

            $('wDays').textContent = totalDays;
            $('wReplied').textContent = replied + '天';
            $('wVisitor').textContent = visitor + '天';
            $('wOptimize').textContent = optimize + '天';
            $('wUpload').textContent = upload + '天';
            $('wSocial').textContent = social + '天';
            $('wOldClient').textContent = oldclient + '天';
            $('wDevelop').textContent = develop + '天';

            // 详情
            let detailHtml = '';
            days.forEach(d => {
                const label = d.date.slice(5);
                if (d.data && d.data.saved) {
                    const tasks = [];
                    if (d.data.replied) tasks.push('✉️');
                    if (d.data.visitor) tasks.push('👤');
                    if (d.data.optimize) tasks.push('🔧');
                    if (d.data.upload) tasks.push('📤');
                    if (d.data.social) tasks.push('📢');
                    if (d.data.oldclient) tasks.push('🤝');
                    if (d.data.develop) tasks.push('🚀');
                    const emoji = tasks.length > 0 ? tasks.join(' ') : '✅';
                    detailHtml += `<div class="day-row"><span>${label}</span><span class="done">${emoji}</span></div>`;
                } else {
                    detailHtml += `<div class="day-row"><span>${label}</span><span class="miss">—</span></div>`;
                }
            });
            $('weekDetail').innerHTML = detailHtml;
            $('weekRange').textContent = `${start} ~ ${end}`;
        }

        // ================================================================
        //  导出 / 导入 / 清除
        // ================================================================
        function exportData() {
            const all = getAllData();
            if (Object.keys(all).length === 0) {
                showToast('暂无数据可导出', '');
                return;
            }
            const blob = new Blob([JSON.stringify(all, null, 2)], { type: 'application/json' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `sop_data_${todayStr()}.json`;
            link.click();
            URL.revokeObjectURL(link.href);
            showToast('📤 导出成功！', 'success');
        }

        function importData(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const data = JSON.parse(e.target.result);
                    if (typeof data !== 'object' || Array.isArray(data)) throw new Error('格式错误');
                    if (confirm('导入将覆盖当前所有数据，确定继续吗？')) {
                        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
                        const date = $('recordDate').value || todayStr();
                        loadDay(date);
                        showToast('✅ 导入成功！', 'success');
                    }
                } catch (err) {
                    showToast('❌ 文件格式无效', '');
                }
            };
            reader.readAsText(file);
            event.target.value = '';
        }

        function clearAll() {
            if (confirm('⚠️ 确定清除所有数据吗？此操作不可撤销！')) {
                localStorage.removeItem(STORAGE_KEY);
                const date = $('recordDate').value || todayStr();
                loadDay(date);
                showToast('已清除所有数据', '');
            }
        }

        let toastTimer = null;

        function showToast(msg, type) {
            const el = document.getElementById('toast');
            el.textContent = msg;
            el.className = 'toast';
            if (type === 'success') el.classList.add('success');
            clearTimeout(toastTimer);
            void el.offsetWidth;
            el.classList.add('show');
            toastTimer = setTimeout(() => el.classList.remove('show'), 2500);
        }
